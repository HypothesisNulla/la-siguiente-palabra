import re
from collections import Counter

# Leer el contenido del archivo
with open('quijote.txt', 'r', encoding='utf-8') as file:
    text = file.read().lower()

# Encontrar todas las palabras que cumplen con la condición
words = re.findall(r'\b[a-záéíóúüñ]{7,}\b', text)

# Contar las ocurrencias de cada palabra
word_counts = Counter(words)

# Encontrar la palabra más común
most_common_word = word_counts.most_common(1)[0][0]
most_common_word
