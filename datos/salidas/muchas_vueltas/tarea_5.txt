import re
from collections import Counter

# Leer el contenido del archivo
with open('quijote.txt', 'r', encoding='utf-8') as file:
    text = file.read().lower()

# Usar una expresión regular para encontrar todas las palabras
words = re.findall(r'\b[a-záéíóúüñ]+\b', text)

# Contar las palabras únicas
unique_words = Counter(words)

# Obtener el número de palabras únicas
num_unique_words = len(unique_words)
num_unique_words
